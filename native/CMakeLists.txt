cmake_minimum_required(VERSION 3.21)
project(board_session LANGUAGES CXX)

if(NOT CMAKE_SYSTEM_NAME MATCHES "^(Linux|Windows)$")
  message(FATAL_ERROR "Unsupported addon target: ${CMAKE_SYSTEM_NAME}")
endif()
set(WOLFSSH_ROOT "" CACHE PATH "Local wolfSSH headers and libraries prefix")
set(WOLFSSL_ROOT "" CACHE PATH "Local wolfSSL headers and libraries prefix")
set(NODE_INCLUDE_DIR "" CACHE PATH "Node.js headers containing node_api.h")
set(ARTIFACT_DIR "" CACHE PATH "Platform artifact directory")
foreach(ENTRY WOLFSSH_INCLUDE WOLFSSL_INCLUDE WOLFSSH_LIBRARY WOLFSSL_LIBRARY)
  unset(${ENTRY} CACHE)
endforeach()
find_path(WOLFSSH_INCLUDE wolfssh/ssh.h PATHS "${WOLFSSH_ROOT}/include" NO_DEFAULT_PATH NO_CACHE REQUIRED)
find_path(WOLFSSL_INCLUDE wolfssl/options.h PATHS "${WOLFSSL_ROOT}/include" NO_DEFAULT_PATH NO_CACHE REQUIRED)
find_library(WOLFSSH_LIBRARY NAMES wolfssh PATHS "${WOLFSSH_ROOT}/lib" NO_DEFAULT_PATH NO_CACHE REQUIRED)
find_library(WOLFSSL_LIBRARY NAMES wolfssl PATHS "${WOLFSSL_ROOT}/lib" NO_DEFAULT_PATH NO_CACHE REQUIRED)
add_library(board_session MODULE addon.cc)
target_compile_features(board_session PRIVATE cxx_std_17)
target_compile_definitions(board_session PRIVATE NAPI_VERSION=8 WOLFSSH_CERTS WOLFSSH_TERM)
target_compile_options(board_session PRIVATE -Wall -Wextra -Werror)
target_include_directories(board_session PRIVATE "${NODE_INCLUDE_DIR}" "${WOLFSSH_INCLUDE}" "${WOLFSSL_INCLUDE}")
target_link_libraries(board_session PRIVATE "${WOLFSSH_LIBRARY}" "${WOLFSSL_LIBRARY}")
set_target_properties(board_session PROPERTIES PREFIX "" SUFFIX ".node" LIBRARY_OUTPUT_DIRECTORY "${ARTIFACT_DIR}")

if(WIN32)
  # Node-API entry points are resolved within addon.cc for Node and Electron.
  target_link_libraries(board_session PRIVATE ws2_32)
  foreach(DLL libwolfssh-18.dll libwolfssl-44.dll)
    add_custom_command(TARGET board_session POST_BUILD
      COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${WOLF_WINDOWS_BIN}/${DLL}" "${ARTIFACT_DIR}/${DLL}")
  endforeach()
  # Capture MinGW's runtime libraries so a Windows machine needs no extra compiler installation.
  foreach(RUNTIME_DLL libstdc++-6.dll libgcc_s_seh-1.dll libwinpthread-1.dll)
    execute_process(COMMAND "${CMAKE_CXX_COMPILER}" "-print-file-name=${RUNTIME_DLL}"
      OUTPUT_VARIABLE RUNTIME_PATH OUTPUT_STRIP_TRAILING_WHITESPACE)
    if(EXISTS "${RUNTIME_PATH}")
      add_custom_command(TARGET board_session POST_BUILD
        COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${RUNTIME_PATH}" "${ARTIFACT_DIR}/${RUNTIME_DLL}")
    endif()
  endforeach()
else()
  target_link_options(board_session PRIVATE "-Wl,--disable-new-dtags")
  set_target_properties(board_session PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN")
  # npm does not preserve SONAME symlinks: copy each actual Linux filename.
  file(GLOB RUNTIME_LIBRARIES "${WOLFSSH_ROOT}/lib/libwolfssh.so*" "${WOLFSSL_ROOT}/lib/libwolfssl.so*")
  foreach(LIBRARY IN LISTS RUNTIME_LIBRARIES)
    get_filename_component(NAME "${LIBRARY}" NAME)
    configure_file("${LIBRARY}" "${ARTIFACT_DIR}/${NAME}" COPYONLY)
  endforeach()
endif()
